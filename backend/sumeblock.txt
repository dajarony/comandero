import logging
from typing import Dict, Any

class MenuParserService:
    def __init__(self, ai, extractor, db):
        self.ai = ai
        self.extractor = extractor
        self.db = db
        self.logger = logging.getLogger("menu_parser")

    async def parse_and_store(self, stream: bytes, rest_id: str) -> Dict[str, Any]:
        """Orquesta la transformación del documento a datos vivos."""
        try:
            # UAF 1: Extracción
            raw_text = self.extractor.get_text(stream)
            
            # UAF 2: Estructuración IA
            structured_data = await self.ai.structure_menu(raw_text)
            
            # UAF 3: Persistencia
            result = await self.db.save_menu(rest_id, structured_data)
            
            self.logger.info(f"Menú procesado para rest_id: {rest_id}")
            return {"success": True, "payload": result}
            
        except Exception as e:
            self.logger.error(f"Error en ingesta de menú: {str(e)}")
            return {"success": False, "error": str(e)}

# Export de instancia para inyección
menu_parser_service = MenuParserService(None, None, None)
