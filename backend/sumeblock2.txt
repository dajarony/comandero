from typing import Dict, Any

class OrderController:
    def __init__(self, service, notifier):
        self.service = service
        self.notifier = notifier

    async def handle_create_order(self, data: Dict[str, Any], user: str) -> Dict[str, Any]:
        """Punto de entrada para la creación de comandas."""
        # Validación rápida de integridad
        if not data.get("items"):
            return {"status": 400, "msg": "Pedido vacío"}

        # Proceso de negocio
        order = await self.service.create(user, data)
        
        # Notificación en tiempo real (STDG broadcast)
        if order:
            await self.notifier.send_to_kitchen(order)
            return {"status": 201, "data": order}
            
        return {"status": 500, "msg": "Error interno"}

# Singleton Controller
order_controller = OrderController(None, None)
