from typing import Dict, Any

class MenuIngestService:
    def __init__(self, ai_provider: Any, db: Any):
        self.ai = ai_provider
        self.db = db
        self.logger = logging.getLogger("service:menu_ingest")

    async def procesar_menu_pdf(self, file_content: bytes) -> Dict[str, Any]:
        """Orquesta la transformación de PDF a menú estructurado."""
        try:
            # 1. Extracción de datos crudos
            raw_text = await self._extraer_texto(file_content)
            
            # 2. Categorización inteligente vía IA
            menu_estructurado = await self._categorizar_con_ia(raw_text)
            
            # 3. Persistencia
            await self.db.save_menu(menu_estructurado)
            
            self.logger.info("Menú procesado exitosamente")
            return {"success": True, "data": menu_estructurado}
            
        except Exception as e:
            self.logger.error(f"Fallo en ingesta: {str(e)}")
            return {"success": False, "error": "Error procesando el documento"}
